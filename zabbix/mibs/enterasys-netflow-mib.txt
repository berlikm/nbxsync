                as(1),

                destinationPrefix(4),

                prefixPort(14),
                bgpNexthopTos(15)
            }

EtsysNetflowOptDataTypes ::= TEXTUAL-CONVENTION
    STATUS  current
    DESCRIPTION
        "The list of optional Netflow version 9 Data FlowSet data 
         that could be supported and enabled on the managed entity.  

         mac(0)    Source and Destination MAC Address
         vlan(1)   Virtual LANs associated with the ingress and egress 
                   interfaces.
         userId(2) Username to user-id bindings will be exported.
        
         higherLayerIfIndex(3)
                   Setting this bit indicates that the interface number 
                   of the highest layer interface will be used.  
                   For example, if a packet ingresses on front panel 
                   interface 11001 and is subsequently routed on a Layer 
                   3 interface with and ifIndex of 3094, the interface 
                   represented in the netflow record will be 3094.  

                   Clearing this bit indicates that the interface number 
                   of the lower layer interface will be used.  In the 
                   previous example the interface represented in the netflow 
                   record would be 11001." 
    SYNTAX  BITS {
                mac(0),
                vlan(1),
                userId(2),
                higherLayerIfIndex(3)
            }

etsysNetflowExportIntfMapTable OBJECT-TYPE
    SYNTAX       SEQUENCE OF EtsysNetflowExportIntfMapEntry
    MAX-ACCESS   not-accessible
    STATUS       current
    DESCRIPTION
        "A table of mappings from the Netflow version 5 export record 
         interfaces to MIB-II ifIndex."
    ::= {  etsysNetflowInterfaceMap 1 }
    
etsysNetflowExportIntfMapEntry OBJECT-TYPE
    SYNTAX       EtsysNetflowExportIntfMapEntry
    MAX-ACCESS   not-accessible
    STATUS       current
    DESCRIPTION
        "An entry containing per interface mapping of Netflow version 
         5 export record interfaces to MIB-II ifIndex."
    INDEX  { etsysNetflowExportIntf }
    ::= { etsysNetflowExportIntfMapTable 1 }

EtsysNetflowExportIntfMapEntry ::=
    SEQUENCE {
        etsysNetflowExportIntf
            Integer32,
        etsysNetflowIfIndex
            InterfaceIndex
    }

etsysNetflowExportIntf OBJECT-TYPE
    SYNTAX       Integer32 (1..65535)
    MAX-ACCESS   not-accessible
    STATUS       current
    DESCRIPTION
        "The interface number used in the 16 bit interface fields of
         Netflow version 5 export records."
    ::= { etsysNetflowExportIntfMapEntry 1 }

etsysNetflowIfIndex OBJECT-TYPE
    SYNTAX       InterfaceIndex
    MAX-ACCESS   read-only
    STATUS       current
    DESCRIPTION
        "The MIB-II ifIndex of the interface bound to 
         etsysNetflowExportIntf."
    ::= { etsysNetflowExportIntfMapEntry 2 }

etsysNetflowIfIndexMapTable OBJECT-TYPE
    SYNTAX       SEQUENCE OF EtsysNetflowIfIndexMapEntry
    MAX-ACCESS   not-accessible
    STATUS       current
    DESCRIPTION
        "A table of mappings from MIB-II ifIndex to Netflow export record 
         interfaces."
    ::= {  etsysNetflowInterfaceMap 2 }

etsysNetflowIfIndexMapEntry OBJECT-TYPE
    SYNTAX       EtsysNetflowIfIndexMapEntry
    MAX-ACCESS   not-accessible
    STATUS       current
    DESCRIPTION
        "An entry containing per interface mapping of MIB-II ifIndex to
         Netflow version 5 export record interfaces."
    INDEX  { ifIndex }
    ::= { etsysNetflowIfIndexMapTable 1 }

EtsysNetflowIfIndexMapEntry ::=
    SEQUENCE {
        etsysNetflowExportInterface
            Integer32
    }

etsysNetflowExportInterface OBJECT-TYPE
    SYNTAX       Integer32 (1..65535)
    MAX-ACCESS   read-only
    STATUS       current
    DESCRIPTION
        "The 16 bit interface number used in the interface fields of
         Netflow version 5 export records that is bound to the specified 
         ifIndex."
    ::= { etsysNetflowIfIndexMapEntry 1 }

-- -------------------------------------------------------------
-- The Netflow Data FlowSet Group
-- -------------------------------------------------------------

etsysNetflowDataFlowSetTable OBJECT-TYPE
    SYNTAX       SEQUENCE OF EtsysNetflowDataFlowSetEntry
    MAX-ACCESS   not-accessible
    STATUS       current
    DESCRIPTION
        "A table containing optional data FlowSet configuration per cache." 
    ::= {  etsysNetflowDataFlowSet 1 }

etsysNetflowDataFlowSetEntry OBJECT-TYPE
    SYNTAX       EtsysNetflowDataFlowSetEntry
    MAX-ACCESS   not-accessible
    STATUS       current
    DESCRIPTION
        "A conceptual row in the etsysNetflowDataFlowSetTable."
    INDEX  { etsysNetflowDataFlowSetCacheType }
    ::= { etsysNetflowDataFlowSetTable 1 }

EtsysNetflowDataFlowSetEntry ::=
    SEQUENCE {
        etsysNetflowDataFlowSetCacheType 
            EtsysNetflowCacheTypes, 
        etsysNetflowOptDataSupported 
            EtsysNetflowOptDataTypes,
        etsysNetflowOptDataEnabled
            EtsysNetflowOptDataTypes
    }

etsysNetflowDataFlowSetCacheType OBJECT-TYPE
    SYNTAX       EtsysNetflowCacheTypes 
    MAX-ACCESS   not-accessible
    STATUS       current
    DESCRIPTION
        "The type of netflow cache.

         NetFlow aggregation maintains one or more extra flow caches
         with different combinations of fields that determine
         which traditional flows are grouped together."
    ::= { etsysNetflowDataFlowSetEntry 1 }

etsysNetflowOptDataSupported OBJECT-TYPE
    SYNTAX       EtsysNetflowOptDataTypes
    MAX-ACCESS   read-only
    STATUS       current
    DESCRIPTION
        "The list of Neflow version 9 data FlowSet optional fields that 
         the managed entity supports."
    ::= { etsysNetflowDataFlowSetEntry 2 }

etsysNetflowOptDataEnabled OBJECT-TYPE
    SYNTAX       EtsysNetflowOptDataTypes
    MAX-ACCESS   read-write
    STATUS       current
    DESCRIPTION
        "The Neflow version 9 data FlowSet optional fields to export. The 
         value of this object is limited to the overlapping set of 
         enumerated bits in the EtsysNetflowOptDataTypes textual 
         convention and the etsysNetflowOptDataSupported object of this 
         conceptual row.  For bits set to 1, the corresponding data will be 
         included in the data FlowSet."
    DEFVAL       { { } }
    ::= { etsysNetflowDataFlowSetEntry 3 }

-- -------------------------------------------------------------
-- The Netflow Module Group
-- -------------------------------------------------------------
etsysNetflowModuleTable OBJECT-TYPE 
    SYNTAX       SEQUENCE OF EtsysNetflowModuleEntry
    MAX-ACCESS   not-accessible
    STATUS       current
    DESCRIPTION
        "A table of per module information and configuration for Netflow."
    ::= {  etsysNetflowModule 1 }
    
etsysNetflowModuleEntry OBJECT-TYPE
    SYNTAX       EtsysNetflowModuleEntry
    MAX-ACCESS   not-accessible
    STATUS       current
    DESCRIPTION
        "An entry containing per module Netflow data.
         Only physical indexes with a entPhysicalClass of module(9)
         are represented in this table.  Furthermore, each entity
         represented in this table must have Netflow resources
         that are separate from every other entity in the table."
    INDEX  { entPhysicalIndex }
    ::= { etsysNetflowModuleTable 1 } 

EtsysNetflowModuleEntry ::= 
    SEQUENCE {
        etsysNetflowMaxRecordExportRate
            Unsigned32,
        etsysNetflowSupportedMaxRecordExportRate
            Unsigned32
    }

etsysNetflowMaxRecordExportRate OBJECT-TYPE
    SYNTAX     Unsigned32 (1..4294967295)
    UNITS      "records per second"
    MAX-ACCESS read-write
    STATUS     current
    DESCRIPTION
        "The maximum number of records the device will export to 
         a configured collector per second.  The actual export rate may
         be less than the configured value and MUST NOT exceed the
         value returned by the etsysNetflowSupportedMaxRecordExportRate
         object.  This object MUST be considered non-volatile and MUST be 
         maintained across entity resets."
    ::= { etsysNetflowModuleEntry 1 }

etsysNetflowSupportedMaxRecordExportRate OBJECT-TYPE
    SYNTAX     Unsigned32
    UNITS      "records per second"
    MAX-ACCESS read-only
    STATUS     current
    DESCRIPTION
        "The supported maximum number of records the device will export to
         a configured collector per second.  The actual export rate may
         be less than the specified value."
    ::= { etsysNetflowModuleEntry 2 }
